# Task: Wallets Module VSA Migration

## Description

Migrate the Wallets module from CQRS/MediatR architecture to Vertical Slice Architecture (VSA) with direct service injection. Consolidate all wallet operation handlers (Increment, Decrement, Transfer, Convert, Release) into a unified `WalletService` while preserving the batch operations infrastructure from Phase 3.5.

## Current Architecture Analysis

**MediatR Command Handlers** (to be consolidated):
- [`IncrementWallet.cs`](src/Modules/XFramework.Wallets/Wallets.Core/Commands/Wallet/IncrementWallet.cs) - Add funds to wallet
- [`DecrementWallet.cs`](src/Modules/XFramework.Wallets/Wallets.Core/Commands/Wallet/DecrementWallet.cs) - Deduct funds from wallet
- [`TransferWallet.cs`](src/Modules/XFramework.Wallets/Wallets.Core/Commands/Wallet/TransferWallet.cs) - Transfer between wallets
- [`ConvertWallet.cs`](src/Modules/XFramework.Wallets/Wallets.Core/Commands/Wallet/ConvertWallet.cs) - Convert wallet types
- [`ReleaseTransaction.cs`](src/Modules/XFramework.Wallets/Wallets.Core/Commands/Wallet/ReleaseTransaction.cs) - Release held transactions
- [`CreateWallet.cs`](src/Modules/XFramework.Wallets/Wallets.Core/Commands/Wallet/CreateWallet.cs) - Create new wallet

**Infrastructure to Preserve**:
- **Batch Operations** (Phase 3.5):
  - [`IBatchWalletService.cs`](src/Modules/XFramework.Wallets/Wallets.Core/Services/IBatchWalletService.cs)
  - [`BatchWalletService.cs`](src/Modules/XFramework.Wallets/Wallets.Core/Services/BatchWalletService.cs)
  - Batch endpoints in `Features/Batch/`
- **Generated Endpoints**: Keep using `[GenerateEndpoints]` for now (Phase 5 will improve this)

## Example Service Method - IncrementBalanceAsync

```csharp
public async Task<Result> IncrementBalanceAsync(
    IncrementWalletRequest request,
    CancellationToken ct = default)
{
    try
    {
        var tenant = await _tenantService.GetTenant(request.Metadata.TenantId);
        
        if (request.TotalAmount <= 0)
        {
            _logger.LogWarning(
                "Invalid increment amount for wallet. WalletId: {WalletId}, Amount: {Amount}",
                request.WalletId, request.TotalAmount);
            return Result.Failure("Invalid increment amount", 400);
        }

        // Fetch wallet
        var wallet = request.WalletTypeId != Guid.Empty
            ? await _dbContext.Set<Wallet>()
                .Where(w => w.TenantId == tenant.Id && 
                           w.WalletTypeId == request.WalletTypeId && 
                           w.CredentialId == request.CredentialId)
                .FirstOrDefaultAsync(ct)
            : await _dbContext.Set<Wallet>()
                .Where(w => w.TenantId == tenant.Id && w.Id == request.WalletId)
                .FirstOrDefaultAsync(ct);

        if (wallet is null)
        {
            // Auto-create wallet if WalletTypeId provided
            if (request.WalletTypeId != Guid.Empty)
            {
                var createResult = await CreateWalletAsync(new CreateWalletRequest
                {
                    CredentialId = request.CredentialId,
                    WalletTypeId = request.WalletTypeId,
                    Balance = 0
                }, ct);
                
                if (!createResult.IsSuccess)
                {
                    return Result.Failure("Error creating wallet", 500);
                }
                
                wallet = createResult.Data;
            }
            else
            {
                return Result.NotFound($"Wallet {request.WalletId} not found");
            }
        }

        // Validate business rules
        if (wallet.MinTransferRule.HasValue && 
            request.TotalAmount < wallet.MinTransferRule.Value)
        {
            return Result.Failure(
                $"Amount must be at least {wallet.MinTransferRule.Value}", 400);
        }

        if (wallet.MaxTransferRule.HasValue && 
            request.TotalAmount > wallet.MaxTransferRule.Value)
        {
            return Result.Failure(
                $"Amount must not exceed {wallet.MaxTransferRule.Value}", 400);
        }

        // Update wallet balance
        var previousBalance = wallet.Balance;
        var previousTotalBalance = wallet.TotalBalance.Value;
        
        if (request.OnHold)
        {
            wallet.CreditOnHoldBalance += request.TotalAmount;
        }
        else
        {
            wallet.Balance += request.TotalAmount;
            wallet.TransferableBalance += request.TotalAmount;
        }

        // Create transaction record
        var transaction = new WalletTransaction
        {
            TenantId = tenant.Id,
            CredentialId = request.CredentialId,
            WalletId = wallet.Id,
            Amount = request.TotalAmount,
            TransactionFee = request.Fee,
            PreviousBalance = previousBalance,
            PreviousTotalBalance = previousTotalBalance,
            RunningBalance = wallet.Balance,
            RunningTotalBalance = wallet.TotalBalance,
            RunningAvailableBalance = wallet.AvailableBalance,
            TransactionType = TransactionType.Credit,
            Held = request.OnHold,
            Released = !request.OnHold,
            ReferenceNumber = string.IsNullOrEmpty(request.ReferenceNumber) 
                ? Guid.NewGuid().ToString() 
                : request.ReferenceNumber,
            Remarks = request.Remarks
        };

        _dbContext.Set<WalletTransaction>().Add(transaction);

        await _dbContext.SaveChangesAsync(ct);
        
        _logger.LogInformation(
            "Wallet balance incremented. WalletId: {WalletId}, Amount: {Amount}",
            wallet.Id, request.TotalAmount);

        return Result.Success();
    }
    catch (DbUpdateConcurrencyException ex)
    {
        _logger.LogError(ex, 
            "Concurrency conflict incrementing wallet. WalletId: {WalletId}",
            request.WalletId);
        return Result.Failure("A concurrency conflict occurred, please try again", 409);
    }
    catch (Exception ex)
    {
        _logger.LogError(ex, 
            "Error incrementing wallet balance. WalletId: {WalletId}",
            request.WalletId);
        return Result.Failure("An error occurred processing your request", 500);
    }
}
```
